package com.example.twsestockapp.ui.main

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.example.twsestockapp.data.DataRepository
import com.example.twsestockapp.data.model.Stock
import com.example.twsestockapp.data.selection.StockFieldSelection
import com.example.twsestockapp.data.selection.StockFieldSelectionRepository
import com.example.twsestockapp.data.selection.StockSelection
import com.example.twsestockapp.data.selection.StockSelectionRepository
import kotlinx.coroutines.flow.Flow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.catch
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.stateIn
import kotlinx.coroutines.launch

/**
 * Default implementation of [MainScreenViewModelContract].
 *
 * Combines:
 *
 * - complete TWSE stock data
 * - persistent ordered stock selections
 * - persistent ordered Pager-field selections
 *
 * into one unified [MainScreenUiState].
 */
class MainScreenViewModel(
    dataRepository: DataRepository,
    private val stockSelectionRepository:
        StockSelectionRepository,
    private val stockFieldSelectionRepository:
        StockFieldSelectionRepository,
) : ViewModel(), MainScreenViewModelContract {

    /**
     * Combined presentation-state stream.
     *
     * The explicit Flow<MainScreenUiState> type is important here.
     *
     * Without it, Kotlin can infer the combine result as
     * Flow<MainScreenUiState.Success>, which would prevent catch from
     * emitting MainScreenUiState.Error.
     */
    private val combinedUiState:
        Flow<MainScreenUiState> =
        combine(
            dataRepository.data,
            stockSelectionRepository.selections,
            stockFieldSelectionRepository.selections,
        ) {
                allStocks,
                selections,
                fieldSelections ->

            createSuccessState(
                allStocks =
                    allStocks,
                selections =
                    selections,
                fieldSelections =
                    fieldSelections,
            )
        }

    /**
     * Complete observable presentation state.
     */
    override val uiState:
        StateFlow<MainScreenUiState> =
        combinedUiState
            .catch { throwable ->
                emit(
                    MainScreenUiState.Error(
                        throwable =
                            throwable,
                    )
                )
            }
            .stateIn(
                scope =
                    viewModelScope,
                started =
                    SharingStarted
                        .WhileSubscribed(5000),
                initialValue =
                    MainScreenUiState.Loading,
            )

    override fun selectStock(
        code: String,
    ) {
        viewModelScope.launch {
            stockSelectionRepository.select(
                code =
                    code,
            )
        }
    }

    override fun unselectStock(
        code: String,
    ) {
        viewModelScope.launch {
            stockSelectionRepository.unselect(
                code =
                    code,
            )
        }
    }

    override fun moveStockForward(
        code: String,
    ) {
        viewModelScope.launch {
            stockSelectionRepository.moveForward(
                code =
                    code,
            )
        }
    }

    override fun moveStockBackward(
        code: String,
    ) {
        viewModelScope.launch {
            stockSelectionRepository.moveBackward(
                code =
                    code,
            )
        }
    }

    override fun clearSelectedStocks() {
        viewModelScope.launch {
            stockSelectionRepository.clear()
        }
    }

    /**
     * Adds one field to the persistent Pager configuration.
     */
    override fun selectStockField(
        fieldKey: String,
    ) {
        viewModelScope.launch {
            stockFieldSelectionRepository.select(
                fieldKey =
                    fieldKey,
            )
        }
    }

    /**
     * Removes one field from the persistent Pager configuration.
     */
    override fun unselectStockField(
        fieldKey: String,
    ) {
        viewModelScope.launch {
            stockFieldSelectionRepository.unselect(
                fieldKey =
                    fieldKey,
            )
        }
    }

    /**
     * Moves one Pager field toward the beginning.
     */
    override fun moveStockFieldForward(
        fieldKey: String,
    ) {
        viewModelScope.launch {
            stockFieldSelectionRepository.moveForward(
                fieldKey =
                    fieldKey,
            )
        }
    }

    /**
     * Moves one Pager field toward the end.
     */
    override fun moveStockFieldBackward(
        fieldKey: String,
    ) {
        viewModelScope.launch {
            stockFieldSelectionRepository.moveBackward(
                fieldKey =
                    fieldKey,
            )
        }
    }

    /**
     * Replaces the complete persistent Pager-field configuration.
     */
    override fun replaceStockFields(
        fieldKeys: List<String>,
    ) {
        viewModelScope.launch {
            stockFieldSelectionRepository.replaceAll(
                fieldKeys =
                    fieldKeys,
            )
        }
    }

    /**
     * Creates presentation state from stock data and both ordered
     * selection collections.
     */
    private fun createSuccessState(
        allStocks: List<Stock>,
        selections: List<StockSelection>,
        fieldSelections: List<StockFieldSelection>,
    ): MainScreenUiState.Success {

        val stocksByCode =
            allStocks.associateBy { stock ->
                stock.Code
            }

        val orderedSelections =
            selections.sortedBy { selection ->
                selection.DisplayOrder
            }

        val visibleStocks =
            orderedSelections.mapNotNull { selection ->
                stocksByCode[
                    selection.Code
                ]
            }

        val orderedFieldSelections =
            fieldSelections.sortedBy { selection ->
                selection.DisplayOrder
            }

        return MainScreenUiState.Success(
            allStocks =
                allStocks,
            visibleStocks =
                visibleStocks,
            selections =
                orderedSelections,
            fieldSelections =
                orderedFieldSelections,
            showSelectionPanel =
                orderedSelections.isEmpty(),
        )
    }
}

/**
 * Route-level entry point for the main stock screen.
 *
 * Production dependencies are obtained from [TWSEStockApplication].
 * UI events are forwarded through [MainScreenViewModelContract].
 */
@Composable
fun MainScreen(
    onItemClick: (NavKey) -> Unit,
    modifier: Modifier = Modifier,
) {
    val application =
        LocalContext.current.applicationContext
            as TWSEStockApplication

    val container =
        application.container

    val viewModel:
        MainScreenViewModelContract =
        viewModel {
            MainScreenViewModel(
                dataRepository =
                    container.dataRepository,
                stockSelectionRepository =
                    container.stockSelectionRepository,
                stockFieldSelectionRepository =
                    container.stockFieldSelectionRepository,
            )
        }

    val uiState by
        viewModel.uiState
            .collectAsStateWithLifecycle()

    var editorRequested by
        rememberSaveable {
            mutableStateOf(false)
        }

    when (val state = uiState) {

        MainScreenUiState.Loading -> {
            MainScreenLoading(
                modifier =
                    modifier,
            )
        }

        is MainScreenUiState.Success -> {
            val showEditor =
                state.showSelectionPanel ||
                    editorRequested

            MainScreenContent(
                state =
                    state,
                showSelectionEditor =
                    showEditor,
                onOpenSelectionEditor = {
                    editorRequested =
                        true
                },
                onCloseSelectionEditor = {
                    if (
                        state.selections.isNotEmpty()
                    ) {
                        editorRequested =
                            false
                    }
                },
                onSelect = { code ->
                    /*
                     * Keep the editor open after the first selection.
                     */
                    editorRequested =
                        true

                    viewModel.selectStock(
                        code =
                            code,
                    )
                },
                onUnselect = { code ->
                    viewModel.unselectStock(
                        code =
                            code,
                    )
                },
                onMoveForward = { code ->
                    viewModel.moveStockForward(
                        code =
                            code,
                    )
                },
                onMoveBackward = { code ->
                    viewModel.moveStockBackward(
                        code =
                            code,
                    )
                },
                modifier =
                    modifier,
            )
        }

        is MainScreenUiState.Error -> {
            MainScreenError(
                throwable =
                    state.throwable,
                modifier =
                    modifier,
            )
        }
    }
}

@OptIn(ExperimentalMaterial3Api::class)
@Composable
internal fun MainScreenContent(
    state: MainScreenUiState.Success,
    showSelectionEditor: Boolean,
    onOpenSelectionEditor: () -> Unit,
    onCloseSelectionEditor: () -> Unit,
    onSelect: (String) -> Unit,
    onUnselect: (String) -> Unit,
    onMoveForward: (String) -> Unit,
    onMoveBackward: (String) -> Unit,
    modifier: Modifier = Modifier,
) {
    Scaffold(
        modifier =
            modifier,
        topBar = {
            CenterAlignedTopAppBar(
                title = {
                    Text(
                        text =
                            if (
                                showSelectionEditor
                            ) {
                                "編輯自選股票"
                            } else {
                                stringResource(
                                    R.string.app_name
                                )
                            }
                    )
                },
                actions = {
                    if (
                        showSelectionEditor
                    ) {
                        TextButton(
                            onClick =
                                onCloseSelectionEditor,
                            enabled =
                                state.selections
                                    .isNotEmpty(),
                        ) {
                            Text(
                                text = "完成"
                            )
                        }
                    }
                },
                colors =
                    TopAppBarDefaults
                        .topAppBarColors(
                            containerColor =
                                MaterialTheme
                                    .colorScheme
                                    .primaryContainer,
                            titleContentColor =
                                MaterialTheme
                                    .colorScheme
                                    .primary,
                        ),
            )
        },
        floatingActionButton = {
            if (
                !showSelectionEditor
            ) {
                FloatingActionButton(
                    onClick =
                        onOpenSelectionEditor,
                    modifier =
                        Modifier.semantics {
                            contentDescription =
                                "編輯自選股票"
                        },
                ) {
                    Text(
                        text = "編輯"
                    )
                }
            }
        },
    ) { innerPadding ->

        if (
            showSelectionEditor
        ) {
            StockSelectionPanel(
                stocks =
                    state.allStocks,
                selections =
                    state.selections,
                onSelect =
                    onSelect,
                onUnselect =
                    onUnselect,
                onMoveForward =
                    onMoveForward,
                onMoveBackward =
                    onMoveBackward,
                modifier =
                    Modifier
                        .fillMaxSize()
                        .padding(

            StockList(
                stocks =
                    state.visibleStocks,
                modifier =
                    Modifier
                        .fillMaxSize()
                        .padding(
                            innerPadding
                        ),
            )
        }
    }
}

private class FakeDataRepository(
    stocks: List<Stock>,
) : DataRepository {

    override val data:
        Flow<List<Stock>> =
        flowOf(
            stocks
        )
}

/**
 * Observable in-memory field-selection repository.
 *
 * This fake preserves continuous one-based DisplayOrder values.
 */
private class FakeStockFieldSelectionRepository(
    initialSelections:
        List<StockFieldSelection>,
) : StockFieldSelectionRepository {

    private val selectionState =
        MutableStateFlow(
            normalizeInitialSelections(
                initialSelections
            )
        )

    override val selections:
        Flow<List<StockFieldSelection>> =
        selectionState

    override suspend fun select(
        fieldKey: String,
    ) {
        if (
            selectionState.value.any { selection ->
                selection.FieldKey ==
                    fieldKey
            }
        ) {
            return
        }

        selectionState.value =
            reindexPreservingOrder(
                selectionState.value +
                    StockFieldSelection(
                        FieldKey =
                            fieldKey,
                        DisplayOrder =
                            selectionState.value
                                .size + 1,
                    )
            )
    }

    override suspend fun unselect(
        fieldKey: String,
    ) {
        selectionState.value =
            reindexPreservingOrder(
                selectionState.value
                    .filterNot { selection ->
                        selection.FieldKey ==
                            fieldKey
                    }
            )
    }

    override suspend fun moveForward(
        fieldKey: String,
    ) {
        val ordered =
            selectionState.value
                .toMutableList()

        val index =
            ordered.indexOfFirst { selection ->
                selection.FieldKey ==
                    fieldKey
            }

        if (
            index <= 0
        ) {
            return
        }

        val previous =
            ordered[index - 1]

        ordered[index - 1] =
            ordered[index]

        ordered[index] =
            previous

        selectionState.value =
            reindexPreservingOrder(
                ordered
            )
    }

    override suspend fun moveBackward(
        fieldKey: String,
    ) {
        val ordered =
            selectionState.value
                .toMutableList()

        val index =
            ordered.indexOfFirst { selection ->
                selection.FieldKey ==
                    fieldKey
            }

        if (
            index < 0 ||
            index >= ordered.lastIndex
        ) {
            return
        }

        val next =
            ordered[index + 1]

        ordered[index + 1] =
            ordered[index]

        ordered[index] =
            next

        selectionState.value =
            reindexPreservingOrder(
                ordered
            )
    }

    override suspend fun replaceAll(
        fieldKeys: List<String>,
    ) {
        selectionState.value =
            fieldKeys
                .distinct()
                .mapIndexed {
                        index,
                        fieldKey ->

                    StockFieldSelection(
                        FieldKey =
                            fieldKey,
                        DisplayOrder =
                            index + 1,
                    )
                }
    }

    override suspend fun clear() {
        selectionState.value =
            emptyList()
    }

    companion object {

        private fun normalizeInitialSelections(
            selections:
                List<StockFieldSelection>,
        ): List<StockFieldSelection> {

            return reindexPreservingOrder(
                selections.sortedBy { selection ->
                    selection.DisplayOrder
                }
            )
        }

        private fun reindexPreservingOrder(
            selections:
                List<StockFieldSelection>,
        ): List<StockFieldSelection> {

            return selections.mapIndexed {
                    index,
                    selection ->

                selection.copy(
                    DisplayOrder =
                        index + 1,
                )
            }
        }
    }
}
